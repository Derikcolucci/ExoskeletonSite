#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>
#include <Adafruit_ADS1X15.h>
#include <Wire.h>
#include <esp_now.h>
#include <WiFi.h>

// --- ESP-NOW WIRELESS SETUP ---
uint8_t broadcastAddress[] = {0x80, 0xF3, 0xDA, 0x5D, 0x89, 0xFC}; // Master MAC
typedef struct struct_message {
    int legID;           
    float kneeAngle;
    float emg1;
    float emg2;
    float emg3;
    float emg4;
} struct_message;

struct_message myData;
esp_now_peer_info_t peerInfo;

// --- I2C BUS DEFINITIONS ---
#define I2C_SDA_IMU 17
#define I2C_SCL_IMU 16
TwoWire I2C_IMU = TwoWire(0);

#define I2C_SDA_ADS 21
#define I2C_SCL_ADS 22
TwoWire I2C_ADS = TwoWire(1);

#define LED_BUILTIN 2

// Sensor Objects
Adafruit_MPU6050 mpuThigh;
Adafruit_MPU6050 mpuShank;
Adafruit_ADS1115 ads; 

// EMG Filtering & Signal States
const int bufferSize = 50;
float vHigh[4] = {0}, vLow[4] = {0};
float smoothVal[4] = {0};
int emgBuffer[4][bufferSize];
int bIndex[4] = {0};
float mvcScale[4] = {500, 500, 500, 500};

// IMU Variables
float thighAngle = 0, shankAngle = 0;
unsigned long lastTime = 0;

// --- REMOTE COMMAND HANDLER ---
void OnDataRecv(const esp_now_recv_info *info, const uint8_t *data, int len) {
    char incomingByte = (char)data[0]; 
    if (incomingByte == 'r') {
        Serial.println("REMOTE RECALIBRATION TRIGGERED!");
        runCalibration();
        lastTime = millis();
    }
}

// --- EMG Processing Function ---
float getProcessedRMS(int sIdx) {
    float rawIn = (float)ads.readADC_SingleEnded(sIdx);
    vHigh[sIdx] = vHigh[sIdx] * 0.95 + rawIn * 0.05;
    float hp = rawIn - vHigh[sIdx];
    vLow[sIdx] = vLow[sIdx] * 0.7 + hp * 0.3;
    emgBuffer[sIdx][bIndex[sIdx]] = (int)vLow[sIdx];
    bIndex[sIdx] = (bIndex[sIdx] + 1) % bufferSize;
    long sqSum = 0;
    for (int i = 0; i < bufferSize; i++) {
        sqSum += (long)emgBuffer[sIdx][i] * (long)emgBuffer[sIdx][i];
    }
    return sqrt(sqSum / (float)bufferSize);
}

// --- Calibration Routine ---
void runCalibration() {
    digitalWrite(LED_BUILTIN, HIGH);
    for(int i=0; i<4; i++) mvcScale[i] = 0.0;
    Serial.println("\n--- STARTING EMG CALIBRATION ---");
    delay(3000); // Relax phase

    Serial.println("\nSTEP 1: Flex THIGH!");
    delay(2000);
    unsigned long tStart = millis();
    while (millis() - tStart < 4000) {
        for (int i = 0; i < 2; i++) {
            float val = getProcessedRMS(i);
            if (val > mvcScale[i]) mvcScale[i] = val;
        }
    }

    Serial.println("\nSTEP 2: Flex HAMSTRINGS!");
    delay(2000);
    tStart = millis();
    while (millis() - tStart < 4000) {
        for (int i = 2; i < 4; i++) {
            float val = getProcessedRMS(i);
            if (val > mvcScale[i]) mvcScale[i] = val;
        }
    }

    for (int i = 0; i < 4; i++) {
        if (mvcScale[i] < 50.0) mvcScale[i] = 1000.0;
    }
    digitalWrite(LED_BUILTIN, LOW);
    Serial.println("--- CALIBRATION COMPLETE ---");
}

void setup() {
    Serial.begin(115200);
    pinMode(LED_BUILTIN, OUTPUT);

    // --- INITIALIZE WIFI & ESP-NOW ---
    WiFi.mode(WIFI_STA);
    if (esp_now_init() != ESP_OK) {
        Serial.println("Error initializing ESP-NOW");
        return;
    }
    
    esp_now_register_recv_cb(OnDataRecv);

    memcpy(peerInfo.peer_addr, broadcastAddress, 6);
    peerInfo.channel = 0;  
    peerInfo.encrypt = false;
    if (esp_now_add_peer(&peerInfo) != ESP_OK){
        Serial.println("Failed to add peer");
        return;
    }

    // Initialize Bus 1: IMUs
    I2C_IMU.begin(I2C_SDA_IMU, I2C_SCL_IMU, 100000); 
    if (!mpuThigh.begin(0x68, &I2C_IMU)) Serial.println("Thigh MPU Fail!");
    if (!mpuShank.begin(0x69, &I2C_IMU)) Serial.println("Shank MPU Fail!");

    // Initialize Bus 2: ADS1115
    I2C_ADS.begin(I2C_SDA_ADS, I2C_SCL_ADS, 400000);
    if (!ads.begin(0x48, &I2C_ADS)) Serial.println("ADS1115 Fail!");
    ads.setDataRate(RATE_ADS1115_860SPS);
    ads.setGain(GAIN_ONE);

    runCalibration();
    lastTime = millis();
}

void loop() {
    // Local serial command check
    if (Serial.available() > 0) {
        char cmd = Serial.read();
        if (cmd == 'r') { 
            while(Serial.available()) Serial.read();
            runCalibration(); 
            lastTime = millis(); 
        }
    }

    unsigned long timestamp = millis();
    float dt = (timestamp - lastTime) / 1000.0;
    lastTime = timestamp;

    sensors_event_t aT, gT, tempT, aS, gS, tempS;
    mpuThigh.getEvent(&aT, &gT, &tempT);
    mpuShank.getEvent(&aS, &gS, &tempS);
    float accelAngleT = atan2(aT.acceleration.y, aT.acceleration.z) * 180 / PI;
    thighAngle = 0.96 * (thighAngle + gT.gyro.x * dt * (180/PI)) + 0.04 * accelAngleT;
    float accelAngleS = atan2(aS.acceleration.y, aS.acceleration.z) * 180 / PI;
    shankAngle = 0.96 * (shankAngle + gS.gyro.x * dt * (180/PI)) + 0.04 * accelAngleS;
    float kneeAngle = thighAngle - shankAngle;

    for (int i = 0; i < 4; i++) {
        float rms = getProcessedRMS(i);
        smoothVal[i] = rms;
    }

    static unsigned long plotLast = 0;
    if (millis() - plotLast > 100) {  // 10 Hz update
        plotLast = millis();
        myData.legID = 1; 
        myData.kneeAngle = kneeAngle;
        myData.emg1 = smoothVal[0];
        myData.emg2 = smoothVal[1];
        myData.emg3 = smoothVal[2];
        myData.emg4 = smoothVal[3];

        esp_err_t result = esp_now_send(broadcastAddress, (uint8_t *) &myData, sizeof(myData));
        if(result != ESP_OK) Serial.println("ESP-NOW Send Error");

        Serial.print("Min:-180,Max:180,");
        Serial.print("Knee_Angle:"); Serial.print(myData.kneeAngle);
        Serial.print(",EMG1:"); Serial.print(myData.emg1);
        Serial.print(",EMG2:"); Serial.print(myData.emg2);
        Serial.print(",EMG3:"); Serial.print(myData.emg3);
        Serial.print(",EMG4:"); Serial.println(myData.emg4);
    }
}
